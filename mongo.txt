# backup
docker compose exec -T db sh -c 'mongodump --archive --username bctool --password phev' > 2022_08_01_bctool.dump

# backup projects only
docker compose exec -T db sh -c 'mongodump --archive --db=bctool --collection=projects --authenticationDatabase=admin --username=bctool --password=phev' > /home/matthew/2024_03_26_bctool_projects.dump

# restore
docker compose exec -T db sh -c 'mongorestore --archive --username bctool --password phev' < 2022_08_01_bctool.dump

# restore projects only
docker compose exec -T db sh -c 'mongorestore --archive --db=bctool --collection=projects --authenticationDatabase=admin --username bctool --password phev' < 2024_03_26_bctool_projects.dump

# cli
docker compose exec db mongosh -u bctool -p
